采用**Socket**编程的方式传递信息
```
import java.io.*;  
import java.net.*;  
public class Sender {  
    public static void main(String[] args) {  
        Socket s = null;  
        BufferedReader br=null;  
        BufferedWriter bw=null;  
        try {  
             s = new Socket(InetAddress.getLocalHost(), 10086);  
            bw = new BufferedWriter(new OutputStreamWriter(s.getOutputStream()));  
            bw.write("hahaha");  
            bw.newLine();  
            bw.flush();  
            s.shutdownOutput();  
            System.out.println("发送端已发送");  
            br=new BufferedReader(new InputStreamReader(s.getInputStream()));  
            System.out.println("接收端发来："+br.readLine());  
  
        } catch (IOException e) {  
            throw new RuntimeException(e);  
        } finally {  
            try {  
                assert s != null;  
                s.close();  
                assert bw != null;  
                bw.close();  
                assert br != null;  
                br.close();  
            } catch (IOException e) {  
                throw new RuntimeException(e);  
            }  
  
        }  
  
  
    }  
}
```

import java.io.*;  
import java.net.ServerSocket;  
import java.net.Socket;  
  
public class reciver {  
    public static void main(String[] args)  {  
        ServerSocket ss= null;  
        Socket s = null;  
        BufferedReader br= null;  
        BufferedWriter bw = null;  
        try {  
            ss = new ServerSocket(10086);  
            s = ss.accept();  
            br = new BufferedReader(new InputStreamReader(s.getInputStream()));  
            String line;  
            System.out.println("接收到的是：");  
            while ((line= br.readLine())!=null) {  
                System.out.println(line);  
            }  
  
            System.out.println("接收完毕，好的长官");  
            bw = new BufferedWriter(new OutputStreamWriter(s.getOutputStream()));  
            bw.write("原味姜水");  
            bw.flush();  
        } catch (IOException e) {  
            throw new RuntimeException(e);  
        } finally {  
            try {  
                assert ss != null;  
                ss.close();  
                assert br != null;  
                br.close();  
                s.close();  
                assert bw != null;  
                bw.close();  
            } catch (IOException e) {  
                throw new RuntimeException(e);  
            }  
        }  
  
  
  
    }  
  
}
```

如果没有接收到信息接收端会在accept和getOutputstream一直停留，不过在没有收到结束标志且与发送端连接再断开后,接收端会跳过getOutputstream运行下一步

收发文件
```
import java.io.*;  
import java.net.ServerSocket;  
import java.net.Socket;  
  
public class reciver {  
    public static void main(String[] args)  {  
        ServerSocket ss= null;  
        Socket s = null;  
        BufferedReader br= null;  
        BufferedWriter bw = null;  
        try {  
            System.out.println("等待中");  
            ss = new ServerSocket(10086);  
            s = ss.accept();  
            br = new BufferedReader(new InputStreamReader(s.getInputStream()));  
            bw = new BufferedWriter(new FileWriter("d:\\hahb.txt"));  
            String line;  
            while ((line= br.readLine())!=null) {  
                bw.write(line);  
                bw.newLine();  
                bw.flush();  
            }  
            System.out.println("下载完成");  
            bw = new BufferedWriter(new OutputStreamWriter(s.getOutputStream()));  
            bw.write("原味姜水");  
            bw.flush();  
        } catch (IOException e) {  
            throw new RuntimeException(e);  
        } finally {  
            try {  
                assert ss != null;  
                ss.close();  
                assert br != null;  
                br.close();  
                assert bw != null;  
                bw.close();  
            } catch (IOException e) {  
                throw new RuntimeException(e);  
            }  
        }  
    }  
  
}
